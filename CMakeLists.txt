cmake_minimum_required(VERSION 3.16)
project(GUIOpenVPNBasiliyWolf VERSION 1.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Network Svg)
add_executable(gui-openvpn-basiliywolf src/main.cpp resources.qrc)
target_link_libraries(gui-openvpn-basiliywolf PRIVATE Qt6::Widgets Qt6::Network Qt6::Svg)
install(TARGETS gui-openvpn-basiliywolf RUNTIME DESTINATION bin)
install(PROGRAMS helper/bw-netns DESTINATION libexec/gui-openvpn-basiliywolf)
install(FILES README.md README.ru.md README.en.md DESTINATION share/doc/gui-openvpn-basiliywolf)
install(FILES gui-openvpn-basiliywolf.desktop DESTINATION share/applications)
install(FILES icons/network98.svg DESTINATION share/icons/hicolor/scalable/apps RENAME gui-openvpn-basiliywolf.svg)
set(CPACK_GENERATOR "DEB")
set(CPACK_PACKAGE_NAME "gui-openvpn-basiliywolf")
set(CPACK_PACKAGE_VERSION "1.0.1")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "BasiliyWolf")
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libqt6widgets6, libqt6network6, libqt6svg6, network-manager, network-manager-openvpn, libsecret-tools, iproute2, nftables, policykit-1, openvpn")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "GUI OpenVPN [aka BasiliyWolf]")
include(CPack)
